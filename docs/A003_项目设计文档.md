// 物理引擎组件
class BubblePhysicsEngine {
  final List<BubbleBody> _bodies = [];
  final Vector2 _gravity = Vector2(0, 9.8);
  final double _airResistance = 0.99;
  
  void update(double dt) {
    for (final body in _bodies) {
      // 应用重力
      body.velocity.add(_gravity * dt);
      
      // 应用空气阻力
      body.velocity.scale(_airResistance);
      
      // 更新位置
      body.position.add(body.velocity * dt);
      
      // 边界检测
      _checkBoundaries(body);
    }
    
    // 碰撞检测
    _checkCollisions();
  }
  
  void _checkCollisions() {
    for (int i = 0; i < _bodies.length; i++) {
      for (int j = i + 1; j < _bodies.length; j++) {
        if (_bodies[i].collidesWith(_bodies[j])) {
          _resolveCollision(_bodies[i], _bodies[j]);
        }
      }
    }
  }
}

// 气泡物理体
class BubbleBody {
  Vector2 position;
  Vector2 velocity;
  double radius;
  double mass;
  
  BubbleBody({
    required this.position,
    required this.velocity,
    required this.radius,
    required this.mass,
  });
  
  bool collidesWith(BubbleBody other) {
    final distance = position.distanceTo(other.position);
    return distance < (radius + other.radius);
  }
}

// 数据加密工具
class SecurityUtils {
  static String encrypt(String data, String key) {
    // 使用AES加密
    final encrypter = Encrypter(AES(Key.fromBase64(key)));
    return encrypter.encrypt(data, iv: IV.fromSecureRandom(16)).base64;
  }
  
  static String decrypt(String encryptedData, String key) {
    // 使用AES解密
    final encrypter = Encrypter(AES(Key.fromBase64(key)));
    return encrypter.decrypt64(encryptedData);
  }
  
  static String hashPassword(String password, String salt) {
    // 使用bcrypt哈希密码
    return BCrypt.hashpw(password, salt);
  }
}

// 网络安全配置
class NetworkSecurity {
  static Dio createSecureClient() {
    return Dio()
      ..interceptors.add(
        InterceptorsWrapper(
          onRequest: (options, handler) {
            // 添加安全头
            options.headers['X-Requested-With'] = 'XMLHttpRequest';
            options.headers['Content-Type'] = 'application/json';
            handler.next(options);
          },
        ),
      );
  }
}

// 平台适配器
class PlatformAdapter {
  static bool get isWeb => kIsWeb;
  static bool get isIOS => !kIsWeb && Platform.isIOS;
  static bool get isAndroid => !kIsWeb && Platform.isAndroid;
  
  static Widget platformSpecificWidget({
    required Widget web,
    required Widget mobile,
    Widget? ios,
    Widget? android,
  }) {
    if (isWeb) return web;
    if (isIOS) return ios ?? mobile;
    if (isAndroid) return android ?? mobile;
    return mobile;
  }
}
